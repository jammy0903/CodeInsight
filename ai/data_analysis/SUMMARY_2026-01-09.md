## 🔧 구현 계획 (Backend)

### 1. MQTT Handler 수정

```python
def on_message(client, userdata, msg):
    data = json.loads(msg.payload)
    device_id = data['header']['src']

    # 버전 감지
    version = detect_hardware_version(device_id)

    if version == 'v1':
        return  # 무시

    elif version == 'v2':
        # 기존 로직
        save_to_influxdb_legacy(data)
        if op == 'fault':
            save_to_f_msg(data)

    elif version == 'v3':
        # 신규 로직
        if not is_duplicate_fault(device_id, data):
            save_to_influxdb_ml(data)
            save_to_f_msg_v3(data)
```

### 2. 중복 제거 로직

```python
def is_duplicate_fault(device_id, data):
    """5초 이내 동일 fault_type=7 체크"""
    cache_key = f"last_fault:{device_id}"
    last_event = cache.get(cache_key)

    if last_event and data['faultType'] == 7:
        time_diff = (now - last_event['time']).seconds
        if time_diff < 5 and last_event['faultType'] == 7:
            return True  # 중복

    cache.set(cache_key, {
        'faultType': data['faultType'],
        'time': now
    })
    return False
```
